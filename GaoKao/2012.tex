        \item 已知函数 $f(x) = a x^{2} + 1$ ($a > 0$), $g(x) = x^{3} + b x$.

\begin{enumerate}[label=(\arabic*)]
    \item 若曲线 $y = f(x)$ 与曲线 $y = g(x)$ 在它们的交点 $(1, c)$ 处具有公共切线，求 $a$, $b$ 的值；
    \item 当 $a^{2} = 4b$ 时，求函数 $f(x) + g(x)$ 的单调区间，并求其在区间 $(-\infty, -1]$ 上的最大值。
\end{enumerate}
        \item 已知曲线 $C: (5-m)x^{2} + (m-2)y^{2} = 8$ ($m \in \mathbb{R}$).

\begin{enumerate}[label=(\arabic*)]
    \item 若曲线 $C$ 是焦点在 $x$ 轴上的椭圆，求 $m$ 的取值范围；
    \item 设 $m=4$，曲线 $C$ 与 $y$ 轴的交点为 $A$, $B$ (点 $A$ 位于点 $B$ 的上方)，直线 $y = kx + 4$ 与曲线 $C$ 交于不同的两点 $M$, $N$，直线 $y = 1$ 与直线 $BM$ 交于点 $G$。求证：$A$, $G$, $N$ 三点共线。
\end{enumerate}
        \item 设 $A$ 是由 $m \times n$ 个实数组成的 $m$ 行 $n$ 列的数表，满足：每个数的绝对值不大于 1，且所有数的和为零。记 $S(m,n)$ 为所有这样的数表构成的集合。对于 $A \in S(m,n)$，记 $r_{i}(A)$ 为 $A$ 的第 $i$ 行各数之和 ($1 \leq i \leq m$)，$c_{j}(A)$ 为 $A$ 的第 $j$ 列各数之和 ($1 \leq j \leq n$)；记 $k(A)$ 为 $\left|r_{1}(A)\right|$, $\left|r_{2}(A)\right|$, $\cdots$, $\left|r_{m}(A)\right|$, $\left|c_{1}(A)\right|$, $\left|c_{2}(A)\right|$, $\cdots$, $\left|c_{n}(A)\right|$ 中的最小值。

\begin{enumerate}[label=(\arabic*)]
    \item 对如下数表 $A$，求 $k(A)$ 的值；
    \begin{table}[H]
        \centering
        \begin{tabular}{|c|c|c|}
        \hline
        $1$ & \(1\) & \(-0.8\) \\ \hline
        \(0.1\)&\(-0.3\)  &\(-1\)  \\ \hline
        \end{tabular}
    \end{table}
    \item 设数表 $A \in S(2,3)$ 形如
       \begin{table}[H]
        \centering
        \begin{tabular}{|c|c|c|}
        \hline
        $1$ & \(1\) & \(c\) \\ \hline
        \(a\)&\(b\)  &\(-1\)  \\ \hline
        \end{tabular}
    \end{table}
    求 $k(A)$ 的最大值；
    \item 给定正整数 $t$，对于所有的 $A \in S(2,2t+1)$，求 $k(A)$ 的最大值。
\end{enumerate}
    \end{enumerate}
\end{description}

    \end{enumerate}
    \item[三、]   
    \begin{enumerate}
      \addtocounter{enumi}{+14}   
        \item 1
        \item 2
        \item 3
        \item 已知函数 $f(x) = a x^{2} + 1$ ($a > 0$), $g(x) = x^{3} + b x$.

\begin{enumerate}[label=(\arabic*)]
    \item 若曲线 $y = f(x)$ 与曲线 $y = g(x)$ 在它们的交点 $(1, c)$ 处具有公共切线，求 $a$, $b$ 的值；
    \item 当 $a = 3$, $b = -9$ 时，若函数 $f(x) + g(x)$ 在区间 $[k, 2]$ 上的最大值为 28，求 $k$ 的取值范围。
\end{enumerate}
        \item 已知椭圆 $C: \frac{x^{2}}{a^{2}} + \frac{y^{2}}{b^{2}} = 1$ ($a > b > 0$) 的一个顶点为 $A(2,0)$，离心率为 $\frac{\sqrt{2}}{2}$，直线 $y = k(x-1)$ 与椭圆 $C$ 交于不同的两点 $M$, $N$.

\begin{enumerate}[label=(\arabic*)]
    \item 求椭圆 $C$ 的方程；
    \item 当 $\triangle AMN$ 的面积为 $\frac{\sqrt{10}}{3}$ 时，求 $k$ 的值。
\end{enumerate}
        \item 设 $A$ 是如下形式的 2 行 3 列的数表，
\begin{center}
\begin{tabular}{|c|c|c|}
\hline
$a$ & $b$ & $c$ \\
\hline
$d$ & $e$ & $f$ \\
\hline
\end{tabular}
\end{center}
满足性质 $P$: $a, b, c, d, e, f \in [-1,1]$, 
且 $a + b + c + d + e + f = 0$.

记 $r_{i}(A)$ 为 $A$ 的第 $i$ 行各数之和 ($i=1,2$), $c_{j}(A)$ 为 $A$ 的第 $j$ 列各数之和 ($j=1,2,3$); 记 $k(A)$ 为 $\left|r_{1}(A)\right|$, $\left|r_{2}(A)\right|$, $\left|c_{1}(A)\right|$, $\left|c_{2}(A)\right|$, $\left|c_{3}(A)\right|$ 中的最小值。

\begin{enumerate}[label=(\arabic*)]
    \item 对如下数表 $A$，求 $k(A)$ 的值；
    \begin{table}[H]
        \centering
        \begin{tabular}{|c|c|c|}
        \hline
        $1$ & \(1\) & \(-0.8\) \\ \hline
        \(0.1\)&\(-0.3\)  &\(-1\)  \\ \hline
        \end{tabular}
    \end{table}
    \item 设数表 $A$ 形如
    \begin{center}
    \begin{tabular}{|c|c|c|}
    \hline
    $1$ & $1$ & $-1-2d$ \\
    \hline
    $d$ & $d$ & $-1$ \\
    \hline
    \end{tabular}
    \end{center}
    其中 $-1 \leq d \leq 0$。求 $k(A)$ 的最大值；
    \item 对所有满足性质 $P$ 的 2 行 3 列的数表 $A$，求 $k(A)$ 的最大值。
\end{enumerate}
    \end{enumerate}
\end{description}
