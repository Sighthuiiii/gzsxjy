\begin{enumerate}[label=(\arabic*)]

\begin{enumerate}[label=(\arabic*)]
    \item 写出一个满足 $a_{1} = a_{5} = 0$，且 $S(A_{5}) > 0$ 的 E 数列 $A_{5}$；
    \item 若 $a_{1} = 12$, $n = 2000$，证明: E 数列 $A_{n}$ 是递增数列的充要条件是 $a_{n} = 2011$；
    \item 对任意给定的整数 $n$ ($n \geqslant 2$), 是否存在首项为 0 的 E 数列 $A_{n}$, 使得 $S(A_{n}) = 0$? 如果存在, 写出一个满足条件的 E 数列 $A_{n}$; 如果不存在, 说明理由。
\end{enumerate}
    \end{enumerate}

    \begin{enumerate}
      \addtocounter{enumi}{+14}   
        \item 1
        \item 2
        \item 3
        \item 已知函数 $f(x) = (x-k) e^{x}$.

\begin{enumerate}[label=(\arabic*)]
    \item 求 $f(x)$ 的单调区间；
    \item 求 $f(x)$ 在区间 $[0,1]$ 上的最小值。
\end{enumerate}
        \item 已知椭圆 $G: \frac{x^{2}}{a^{2}} + \frac{y^{2}}{b^{2}} = 1$ ($a > b > 0$) 的离心率为 $\frac{\sqrt{6}}{3}$，右焦点为 $(2\sqrt{2}, 0)$，斜率为 $1$ 的直线 $l$ 与椭圆 $G$ 交于 $A$, $B$ 两点，以 $AB$ 为底边作等腰三角形，顶点为 $P(-3,2)$。

\begin{enumerate}[label=(\arabic*)]
    \item 求椭圆 $G$ 的方程；
    \item 求 $\triangle PAB$ 的面积。
\end{enumerate}
        \item 若数列 $A_{n}: a_{1}, a_{2}, \cdots, a_{n} (n \geqslant 2)$ 满足 
        $\left| a_{n+1} - a_{k} \right| = 1$ $(k=1,2,\cdots,n-1)$，
        则称 $A_{n}$ 为 $E$ 数列。记 $S(A_{n}) = a_{1} + a_{2} + \cdots + a_{n}$。

\begin{enumerate}[label=(\arabic*)]
    \item 写出一个 $E$ 数列 $A_{5}$ 满足 $a_{1} = a_{3} = 0$；
    \item 若 $a_{1} = 12$, $n = 2000$，证明：E 数列 $A_{n}$ 是递增数列的充要条件是 $a_{n} = 2011$；
    \item 在 $a_{1} = 4$ 的 $E$ 数列 $A_{n}$ 中，求使得 $S(A_{n}) = 0$ 成立的 $n$ 的最小值。
\end{enumerate}
    \end{enumerate}
\end{description}
